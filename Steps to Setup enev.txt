cd D:\Project_3

# 1) Create a virtual environment
py -3.12 -m venv .venv

# 2) Activate it
.\.venv\Scripts\Activate.ps1

# 3) Upgrade pip
python -m pip install --upgrade pip

# 4) Install required packages
pip install langchain-core langchain-openai langgraph streamlit

# 5) Set your OpenAI key
$env:OPENAI_API_KEY = "your_actual_openai_api_key_here"

# 6) Confirm the project modules import
python -c "import tools, agent; print('IMPORT_OK')"

# 7) Run the agent directly
python agent.py

# OR run the Streamlit app
streamlit run app.py
